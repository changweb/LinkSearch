// 2. สร้างตัวเลือกวันที่ใน Dropdown อัตโนมัติ (แปลงเป็นเวลาไทยก่อน) function populateDateDropdown() { const select = document.getElementById('dateFilter'); const uniqueDates = new Set(); dataArray.forEach(item => { if(item.date) { let d = new Date(item.date); if (!isNaN(d)) { // แปลงเป็น YYYY-MM-DD แบบเวลาไทย let localDate = d.toLocaleDateString("en-CA", { timeZone: "Asia/Bangkok" }); uniqueDates.add(localDate); } } }); const sortedDates = Array.from(uniqueDates).sort().reverse(); let optionsHtml = ''; sortedDates.forEach(date => { optionsHtml += ``; }); select.innerHTML = optionsHtml; } // 3. กรองและแสดงผลข้อมูลตาราง function filterData() { const dateFilter = document.getElementById('dateFilter').value; const searchTxt = document.getElementById('searchInput').value.toLowerCase(); let filtered = dataArray; // กรองตามวันที่ (แปลงเวลาไทยก่อนเทียบ) if (dateFilter !== 'all') { filtered = filtered.filter(item => { if(!item.date) return false; let d = new Date(item.date); let localDate = !isNaN(d) ? d.toLocaleDateString("en-CA", { timeZone: "Asia/Bangkok" }) : item.date; return localDate.includes(dateFilter); }); } // กรองตามคำค้นหา if (searchTxt !== '') { filtered = filtered.filter(item => (item.linkname && item.linkname.toLowerCase().includes(searchTxt)) || (item.link && item.link.toLowerCase().includes(searchTxt)) ); } renderTable(filtered); } function renderTable(items) { const tbody = document.getElementById('tableBody'); if (items.length === 0) { tbody.innerHTML = 'ไม่พบข้อมูล'; return; } tbody.innerHTML = items.map(item => { // แปลงวันที่แสดงผลให้เป็นเวลาไทย และเป็นรูปแบบที่อ่านง่าย (เช่น 8 ก.ค. 2569) let displayDate = '-'; if (item.date) { let d = new Date(item.date); displayDate = !isNaN(d) ? d.toLocaleDateString("th-TH", { timeZone: "Asia/Bangkok", year: 'numeric', month: 'short', day: 'numeric' }) : item.date.split('T')[0]; } let displayLink = item.link && item.link.includes('http') ? `เปิดดูคลิป` : item.link; return ` ${displayDate} ${item.linkname || '-'} ${displayLink} `; }).join(''); lucide.createIcons(); }
